      function loadSections() {
        if (!database) return;
        database.ref('sections').on('value', (snapshot) => {
          sectionsData = [];
          snapshot.forEach(child => {
            sectionsData.push({ id: child.key, ...child.val() });
          });
          console.log('sections loaded', sectionsData.length);
          // refresh section radio group if visible
          const currentBranch = document.querySelector('input[name="branche"]:checked')?.value;
          if (currentBranch) handleSecretaryBranchSelection(currentBranch);
        });
      }
